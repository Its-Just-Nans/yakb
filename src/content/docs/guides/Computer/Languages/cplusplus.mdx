// 13. COPY / MOVE SEMANTICS
// ============================================================

class Buffer
{
private:
    size_t size;
    int* data;

public:
    explicit Buffer(size_t size)
        : size(size),
          data(new int[size])
    {
        cout << "Constructor\n";
    }

    ~Buffer()
    {
        delete[] data;
        cout << "Destructor\n";
    }

    // Copy constructor
    Buffer(const Buffer& other)
        : size(other.size),
          data(new int[other.size])
    {
        copy(
            other.data,
            other.data + size,
            data
        );

        cout << "Copy constructor\n";
    }

    // Copy assignment
    Buffer& operator=(const Buffer& other)
    {
        if (this != &other)
        {
            int* newData = new int[other.size];

            copy(
                other.data,
                other.data + other.size,
                newData
            );

            delete[] data;

            data = newData;
            size = other.size;
        }

        cout << "Copy assignment\n";

        return *this;
    }

    // Move constructor
    Buffer(Buffer&& other) noexcept
        : size(other.size),
          data(other.data)
    {
        other.size = 0;
        other.data = nullptr;

        cout << "Move constructor\n";
    }

    // Move assignment
    Buffer& operator=(Buffer&& other) noexcept
    {
        if (this != &other)
        {
            delete[] data;

            size = other.size;
            data = other.data;

            other.size = 0;
            other.data = nullptr;
        }

        cout << "Move assignment\n";

        return *this;
    }
};

void copyMoveDemo()
{
    cout << "\n=== 13. COPY / MOVE ===\n";

    Buffer a(100);

    Buffer b = a;

    Buffer c = move(a);

optional<int> findValue(
    const vector<int>& values,
    int target)
{
    for (int value : values)
    {
        if (value == target)
            return value;
    }

    return nullopt;
}

void optionalDemo()
{
    cout << "\n=== 29. OPTIONAL ===\n";

    vector<int> values{
        10, 20, 30
    };

    auto result = findValue(values, 20);

    if (result)
        cout << "Found: "
             << *result
             << '\n';

    auto missing = findValue(values, 100);

    cout << "Missing: "
         << missing.value_or(-1)
         << '\n';
}

mutex cvMutex;
condition_variable cv;
bool ready = false;

void waitingThread()
{
    unique_lock<mutex> lock(cvMutex);

    cv.wait(
        lock,
        []
        {
            return ready;
        }
    );

    cout << "Worker received notification\n";
}

void conditionVariableDemo()
{
    cout << "\n=== 50. CONDITION VARIABLE ===\n";

    thread workerThread(
        waitingThread
    );

    this_thread::sleep_for(
        chrono::milliseconds(100)
    );

    {
        lock_guard<mutex> lock(cvMutex);

        ready = true;
    }

    cv.notify_one();

    workerThread.join();
}
